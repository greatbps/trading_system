✅ 왜 ATR(평균진폭지표, Average True Range)을 쓰면 좋은가?

시장 상황에 유연하게 대응

고정 % 손절/익절은 변동성이 큰 장에서는 너무 일찍 잘리고, 변동성이 낮은 장에서는 너무 늦게 반응할 수 있음.

ATR 기반 트레일링은 종목의 실제 변동성 크기를 반영해서 스탑 거리를 조절합니다.

가짜 신호(노이즈) 방지

예를 들어 3분봉에서 단순히 5EMA만 기준 삼으면 작은 흔들림에도 청산될 수 있어요.

ATR을 곱한 트레일링(예: 1.5 * ATR)을 쓰면 “진짜 추세가 꺾였을 때만 청산”되는 확률이 커집니다.

익절 후 수익 극대화

+6% 달성 시 전부 매도하는 대신, ATR 트레일링을 걸어두면 추세가 길게 이어질 때 더 큰 수익을 뽑아낼 수 있어요.

✅ 추천 구조 (Python 백테스트/실전 구현 로직)
# 포지션 진입 시
entry_price = current_price
stop_loss = entry_price * 0.97   # -3% 하드스탑
target_price = entry_price * 1.06  # +6% 소프트 타깃
trailing_stop = None
highest_price = entry_price

# 매 3분봉마다 업데이트
for candle in candles_3m:
    close = candle['close']
    atr = ATR(candles_3m, period=14)  # 14기간 ATR
    
    # 손절 조건
    if close <= stop_loss:
        sell_all("hard stop loss")
        break

    # 익절 조건 진입
    if close >= target_price and trailing_stop is None:
        # 부분익절 실행 가능 (예: 50%)
        sell_half("partial take profit")
        highest_price = close
        trailing_stop = close - 1.5 * atr  # 초기 트레일링 설정
    
    # 익절 후 트레일링 적용
    if trailing_stop is not None:
        highest_price = max(highest_price, close)
        trailing_stop = highest_price - 1.5 * atr
        
        if close <= trailing_stop:
            sell_all("ATR trailing stop triggered")
            break
    
    # 추가 조건: EMA5 하향 이탈 + 볼륨 증가 시 강제 청산
    ema5 = EMA(candles_3m, period=5)
    avg_vol = np.mean([c['volume'] for c in candles_3m[-20:]])
    if close >= target_price and close < ema5 and candle['volume'] > 1.2 * avg_vol:
        sell_all("EMA breakdown with volume")
        break

1️⃣ 그리드 서치(Grid Search)

import itertools

atr_periods = [10, 14, 20]
atr_multipliers = [1.0, 1.5, 2.0, 2.5]
stop_losses = [0.97, 0.98]
take_profits = [1.05, 1.06, 1.08]

best_result = None
best_params = None

for p, k, sl, tp in itertools.product(atr_periods, atr_multipliers, stop_losses, take_profits):
    result = backtest_strategy(data, atr_period=p, atr_mult=k, sl=sl, tp=tp)
    
    if best_result is None or result['sharpe'] > best_result['sharpe']:
        best_result = result
        best_params = (p, k, sl, tp)

print("최적 파라미터:", best_params, "성과:", best_result)

2️⃣ 변동성 기반 동적 조정 (Adaptive ATR)

종목의 **평균 변동성(표준편차 or ATR 레벨)**에 따라 자동으로 k 값 조정.

변동성이 낮으면 작은 k(예: 1.2), 변동성이 크면 큰 k(예: 2.0).

예시 로직:

volatility = ATR(data, 14) / data['close']  # 상대 ATR (% 단위)
if volatility < 0.01:   # 변동성 낮음
    atr_mult = 1.2
elif volatility < 0.02:
    atr_mult = 1.5
else:                   # 변동성 높음
    atr_mult = 2.0

3️⃣ Bayesian Optimization / Genetic Algorithm

Grid Search는 느릴 수 있으니, hyperopt, optuna, scikit-optimize 같은 라이브러리를 활용하면 자동으로 최적 파라미터 탐색 가능.

수백~수천 조합을 효율적으로 탐색해 가장 높은 기대수익률 or 샤프비율을 주는 값을 찾아줍니다.

from optuna import create_study

def objective(trial):
    p = trial.suggest_int("atr_period", 10, 30)
    k = trial.suggest_float("atr_mult", 1.0, 3.0, step=0.1)
    sl = trial.suggest_float("stop_loss", 0.95, 0.99, step=0.01)
    tp = trial.suggest_float("take_profit", 1.03, 1.10, step=0.01)

    result = backtest_strategy(data, atr_period=p, atr_mult=k, sl=sl, tp=tp)
    return result['sharpe']

study = create_study(direction="maximize")
study.optimize(objective, n_trials=100)
print("Best params:", study.best_params)

import os
import warnings

# -------------------------
# Strategy: ATR trailing + fixed TP/SL + 부분 익절(예시 Entry signal: SMA crossover)
# -------------------------
class ATRTrailStrategy(bt.Strategy):
    params = dict(
        atr_period=14,
        atr_mult=1.5,
        stop_loss_pct=0.97,   # -3% hard stop (entry * stop_loss_pct)
        take_profit_pct=1.06, # +6% soft target (entry * take_profit_pct)
        partial_take_pct=0.4, # 부분 익절 비율 (예: 40%)
        partial_tp_pct=1.04,  # 부분익절 가격(예: +4%)
        use_partial=True,
        verbose=False,
    )

    def __init__(self):
        self.atr = bt.ind.ATR(self.data, period=self.p.atr_period)
        self.ema5 = bt.ind.EMA(self.data.close, period=5)
        self.order = None
        self.entry_price = None
        self.highest = None
        self.trailing_stop = None
        self.position_size = 0

        # 간단 예시 엔트리 지표: 20SMA를 상향돌파하면 매수 (사용자는 이 부분을 커스터마이즈)
        self.sma20 = bt.ind.SimpleMovingAverage(self.data.close, period=20)
        self.crossup = bt.ind.CrossOver(self.data.close, self.sma20)

    def log(self, txt):
        if self.p.verbose:
            dt = self.datas[0].datetime.datetime(0)
            print(f"{dt.isoformat()} - {txt}")

    def next(self):
        # 이미 주문 대기 중이면 패스
        if self.order:
            return

        # 진입 조건 (여기서는 예시로 SMA20 상향돌파)
        if not self.position and self.crossup[0] > 0:
            size = self.broker.getcash() / self.data.close[0] * 0.1  # 예: 자금의 10% 사용 (실사용 시 수정 권장)
            size = int(size)
            if size <= 0:
                return
            self.order = self.buy(size=size)
            self.log(f"BUY CREATED price={self.data.close[0]:.2f} size={size}")

        # 포지션 관리
        if self.position:
            close = self.data.close[0]
            # 최초 진입 직후 정보 세팅
            if self.entry_price is None:
                self.entry_price = self.position.price
                self.highest = close
                self.trailing_stop = None
                self.position_size = self.position.size
                self.log(f"ENTERED @ {self.entry_price:.4f}")

            # update highest price since entry
            self.highest = max(self.highest, close)

            # 초기 하드 stop
            hard_stop = self.entry_price * self.p.stop_loss_pct
            # soft TP target
            tp_price = self.entry_price * self.p.take_profit_pct

            # partial TP
            if self.p.use_partial and close >= self.p.partial_tp_pct * self.entry_price:
                # 부분 익절: 단 한 번만 수행 (여기서는 잔여 포지션이 있으면 부분 매도)
                if self.position_size > 0:
                    amount = int(self.position_size * self.p.partial_take_pct)
                    if amount > 0:
                        self.sell(size=amount)
                        self.position_size -= amount
                        self.log(f"PARTIAL TAKE: sold {amount} @ {close:.4f}")

            # TP에 도달하면 trailing 시작 (초기)
            if close >= tp_price and self.trailing_stop is None:
                self.trailing_stop = self.highest - (self.p.atr_mult * self.atr[0])
                self.log(f"TP reached, set initial trailing_stop={self.trailing_stop:.4f}")

            # trailing stop update
            if self.trailing_stop is not None:
                new_trailing = self.highest - (self.p.atr_mult * self.atr[0])
                if new_trailing > self.trailing_stop:
                    self.trailing_stop = new_trailing
                    # self.log(f"trailing updated to {self.trailing_stop:.4f}")

            # EMA5 + volume 조건으로 즉시 청산(옵션): 여기서는 단순 EMA5 하향 이탈 시 청산
            if close < self.ema5[0] and close >= tp_price:
                # 포지션 전량 청산
                self.close()
                self.log("EMA5 breakdown after TP -> CLOSE ALL")
                self._reset_entry_state()
                return

            # trailing stop hit
            if self.trailing_stop is not None and close <= self.trailing_stop:
                self.close()
                self.log("ATR trailing stop hit -> CLOSE ALL")
                self._reset_entry_state()
                return

            # hard stop hit
            if close <= hard_stop:
                self.close()
                self.log("Hard stop hit -> CLOSE ALL")
                self._reset_entry_state()
                return

    def notify_order(self, order):
        if order.status in [order.Completed]:
            if order.isbuy():
                self.log(f"BUY EXECUTED price={order.executed.price:.4f}, size={order.executed.size}")
            elif order.issell():
                self.log(f"SELL EXECUTED price={order.executed.price:.4f}, size={order.executed.size}")
            self.order = None
        elif order.status in [order.Canceled, order.Rejected]:
            self.log("Order Canceled/Rejected")
            self.order = None

    def notify_trade(self, trade):
        if trade.isclosed:
            self.log(f"TRADE P/L gross {trade.pnl:.4f}, net {trade.pnlcomm:.4f}")

    def _reset_entry_state(self):
        self.entry_price = None
        self.highest = None
        self.trailing_stop = None
        self.position_size = 0

# Optuna objective
# -------------------------
def objective(trial, datafeed, cash=100000.0, commission=0.0005):
    # search space
    atr_period = trial.suggest_int("atr_period", 8, 24)
    atr_mult = trial.suggest_float("atr_mult", 1.0, 3.0, step=0.1)
    stop_loss_pct = trial.suggest_float("stop_loss_pct", 0.94, 0.99, step=0.01)  # 0.94 => -6%
    take_profit_pct = trial.suggest_float("take_profit_pct", 1.03, 1.12, step=0.01)
    partial_take_pct = trial.suggest_float("partial_take_pct", 0.0, 0.6, step=0.1)
    partial_tp_pct = trial.suggest_float("partial_tp_pct", 1.02, 1.08, step=0.01)
    use_partial = partial_take_pct > 0.0

    cerebro = bt.Cerebro(stdstats=False)
    cerebro.broker.setcash(cash)
    cerebro.broker.setcommission(commission=commission)

    # Add data
    cerebro.adddata(datafeed)

    # Add strategy with suggested params
    cerebro.addstrategy(ATRTrailStrategy,
                        atr_period=atr_period,
                        atr_mult=atr_mult,
                        stop_loss_pct=stop_loss_pct,
                        take_profit_pct=take_profit_pct,
                        partial_take_pct=partial_take_pct,
                        partial_tp_pct=partial_tp_pct,
                        use_partial=use_partial,
                        verbose=False)

    # run
    try:
        results = cerebro.run()
    except Exception as e:
        # if strategy errors (e.g., size 0), return poor score
        return -1e6

    final_value = cerebro.broker.getvalue()
    pnl = final_value - cash

    # 간단 샤프(대략): 일별 리턴 사용하려면 일별 포트폴리오 value series 필요 -> here we use pnl to rank
    # To be more rigorous, implement a broker value analyzer or resample data for daily value series
    return pnl  # maximize total PnL (사용자 원하면 샤프로 변경)

# Main CLI flow
# -------------------------
def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--data", required=True, help="OHLCV CSV file path (datetime must be column name)")
    parser.add_argument("--trials", type=int, default=50, help="Optuna trials")
    parser.add_argument("--out", default="optuna_study.db", help="Optuna sqlite DB file")
    parser.add_argument("--cash", type=float, default=100000.0)
    args = parser.parse_args()

    csv_path = args.data
    if not os.path.exists(csv_path):
        raise FileNotFoundError(f"{csv_path} not found")

    # CSV load
    df = pd.read_csv(csv_path, parse_dates=['datetime'])
    df = df.sort_values('datetime').set_index('datetime')
    # backtrader feed expects datetime index and OHLCV columns
    df_bt = df.rename(columns={
        'open': 'open', 'high': 'high', 'low': 'low', 'close': 'close', 'volume': 'volume'
    })[['open', 'high', 'low', 'close', 'volume']].copy()
    df_bt['openinterest'] = 0
    datafeed = bt.feeds.PandasData(dataname=df_bt)

    # Optuna study
    storage = f"sqlite:///{args.out}"
    study = optuna.create_study(direction="maximize", study_name="backtest_study", storage=storage, load_if_exists=True)

    def _objective(trial):
        return objective(trial, datafeed, cash=args.cash)

    print(f"Start optimization: trials={args.trials}, storage={storage}")
    study.optimize(_objective, n_trials=args.trials)

    print("Best trial:")
    print(study.best_trial.params)
    print("Value (PnL):", study.best_value)

    # After best params found, run final backtest and show result + plot
    best = study.best_trial.params
    cerebro = bt.Cerebro(stdstats=False)
    cerebro.broker.setcash(args.cash)
    cerebro.adddata(datafeed)
    cerebro.addstrategy(ATRTrailStrategy,
                        atr_period=best['atr_period'],
                        atr_mult=best['atr_mult'],
                        stop_loss_pct=best['stop_loss_pct'],
                        take_profit_pct=best['take_profit_pct'],
                        partial_take_pct=best.get('partial_take_pct', 0.0),
                        partial_tp_pct=best.get('partial_tp_pct', 1.02),
                        use_partial=best.get('partial_take_pct', 0.0) > 0.0,
                        verbose=True)
    cerebro.run()
    print("Final portfolio value:", cerebro.broker.getvalue())
    try:
        cerebro.plot()
    except Exception:
        print("플롯을 표시할 수 없습니다. (headless 환경일 수 있음)")

if __name__ == "__main__":
    main()
